Juntar e contar: JOIN, agregações e o resumo dos kits

Até aqui, cada consulta lia uma tabela por vez. Hoje a coordenação pede um resumo: quantos itens e unidades cada kit tem, inclusive os vazios, e quais itens de todos os kits estão acabando. Você junta kit e item com JOIN, conta e soma com COUNT e SUM, agrupa com GROUP BY e mostra tudo numa janela de resumo. No fim, depura um resumo herdado que contava um item no kit vazio, juntava kits de mesmo nome e repetia cada item com todos os kits.

A aula continua na janela da A16, no DadosMySQL e no MySQL da A08. O tutorial usa o SQLite da A06, em memória, porque o que se vê é o mesmo; as diferenças do MySQL estão no texto. Use dados fictícios.

JOIN: quem fica quando o outro lado não tem par ↗

Um JOIN junta cada linha de uma tabela com as linhas da outra que satisfazem a condição do ON: item.kit_id = kit.id liga cada item ao seu kit. Sem condição, o banco junta tudo com tudo, e cada item aparece uma vez para cada kit; é o relato 3.

No INNER JOIN, que é o JOIN sem adjetivo, só ficam as linhas que têm par dos dois lados: um kit sem itens não aparece. Para a lista de itens acabando, é o certo: o que interessa são itens, e todo item tem kit. Para o resumo, não: o kit vazio precisa estar lá, com zero. É o LEFT JOIN: todas as linhas da tabela da esquerda ficam, e, quando não há par, as colunas da direita vêm NULL.

Contar e somar sem enganar ↗

GROUP BY junta as linhas do JOIN em grupos, e cada grupo vira uma linha do resultado; COUNT e SUM calculam por grupo. O detalhe está no kit vazio: o LEFT JOIN devolve para ele uma linha, com o lado do item todo NULL. COUNT(*) conta linhas, então conta essa também: o kit vazio fica com 1, o relato 1. COUNT(i.id) conta só os valores que não são NULL: zero.

Com SUM é parecido: a soma de nenhum valor é NULL, e não 0. COALESCE(SUM(i.quantidade), 0) devolve o primeiro valor que não for NULL: a soma, ou 0. No MySQL, SUM de uma coluna INT volta como DECIMAL, e o conector entrega um Decimal do Python; resumo_kits converte com int(), para a janela e os testes receberem números inteiros. No SQLite do tutorial, a soma já volta int.

Agrupe pela chave. Com GROUP BY k.nome, dois kits com o mesmo nome viram um grupo só, e a soma junta os estoques: o relato 2. Com GROUP BY k.id, k.nome, cada kit é um grupo, mesmo com nomes iguais. O MySQL 8.4, com o ONLY_FULL_GROUP_BY ligado de fábrica, recusa um SELECT com coluna que não está no GROUP BY nem dentro de uma agregação; por isso o resumo herdado precisou de MIN(k.id).

import unittest

import customtkinter as ctk

from kits.resumo import JanelaResumo
from tests.test_janela import FORA, KITS, BancoDeMentira, fechar


class TestResumo(unittest.TestCase):
    def abrir(self, banco=None):
        ctk.set_appearance_mode('light')
        self.raiz = ctk.CTk()
        self.raiz.withdraw()
        self.addCleanup(fechar, self.raiz)
        self.banco = banco or BancoDeMentira(kits=KITS + [(3, 'Almoxarifado')])
        resumo = JanelaResumo(self.raiz, self.banco)
        resumo.update()
        return resumo

    def linhas(self, tabela):
        return [tuple(tabela.item(i, 'values')) for i in tabela.get_children()]

    def test_cada_kit_com_itens_e_unidades_inclusive_o_vazio(self):
        resumo = self.abrir()
        self.assertEqual(self.linhas(resumo.kits), [('Almoxarifado', '0', '0'), ('Treinamento Redes', '2', '5'),

        self.assertEqual(resumo.total.cget('text'), '3 kits · 3 itens · 10 unidades')

    def test_itens_abaixo_de_3_por_padrao(self):

def fechar(janela):
    # A17: as janelas filhas (o resumo) primeiro. Cada uma leva junto o que agendou; se a tarefa dela fosse
    # cancelada pela janela principal, o destroy da filha tentaria apagar um comando que já não existe.
    for filha in [w for w in janela.winfo_children() if isinstance(w, tkinter.Toplevel)]:
        filha.destroy()
    # O CustomTkinter agenda tarefas com after (o ícone da barra de título, por exemplo).
    # Cancelar antes de destruir evita avisos de "invalid command name" no fim do teste.
    for tarefa in janela.tk.call('after', 'info'):
        janela.after_cancel(tarefa)
    janela.destroy()


class BancoDeMentira:
    """Guarda kits e itens em listas e registra o que a janela pediu. falhas: {nome do método: exceção}."""

    def __init__(self, kits=KITS, itens=ITENS, resposta=True, nome_novo=None):
        self.kits = list(kits)
        self.itens = copy.deepcopy(itens)
        self.resposta = resposta
        self.nome_novo = nome_novo
        self.pedidos_de_nome = []
        self.kits_criados = []
        self.falhas = {}
        self.pedidos = []
        self.limites = []
        self.consultas = []
        self.entregues = []
        self.removidos = []
        self.perguntas = []
        self.editados = []
        self.kits_removidos = []

    def _talvez_falhar(self, nome):
        if nome in self.falhas:
            raise self.falhas[nome]

    def listar_kits(self):
        self._talvez_falhar('listar_kits')
        return list(self.kits)

    def listar_itens(self, kit_id, trecho='', ordem='nome', descendente=False):
        """Busca e ordena como o MySQL faria com estes dados: trecho em qualquer parte do nome, sem
        diferenciar maiúsculas; empate pelo id. Guarda cada pedido em consultas."""
        self._talvez_falhar('listar_itens')
        self.pedidos.append(kit_id)
        self.consultas.append((kit_id, trecho, ordem, descendente))
        itens = [i for i in self.itens.get(kit_id, []) if trecho.casefold() in i[1].casefold()]
        coluna = {'nome': 1, 'quantidade': 2}[ordem]
        itens.sort(key=lambda i: i[0])
        itens.sort(key=lambda i: i[coluna].casefold() if coluna == 1 else i[coluna], reverse=descendente)
        return itens

    def resumo_kits(self):
        """Conta e soma como o resumo_kits do repositório: todos os kits, em ordem de nome e id."""
        self._talvez_falhar('resumo_kits')
        resumo = [(k, nome, len(self.itens.get(k, [])), sum(q for _, _, q in self.itens.get(k, [])))
                  for k, nome in self.kits]
        return sorted(resumo, key=lambda linha: (linha[1].casefold(), linha[0]))

    def itens_abaixo(self, limite):
        self._talvez_falhar('itens_abaixo')
        self.limites.append(limite)
        nomes = dict(self.kits)
        itens = [(nomes[k], nome, q) for k, lista in self.itens.items() for _, nome, q in lista if q < limite]
        return sorted(itens, key=lambda linha: (linha[2], linha[0].casefold(), linha[1].casefold()))

    def criar_kit(self, nome):
        self._talvez_falhar('criar_kit')
        kit_id = max([k for k, _ in self.kits], default=0) + 1
        self.kits.append((kit_id, nome.strip()))
        self.kits_criados.append(nome)
        return kit_id

    def adicionar(self, kit_id, item):
        self._talvez_falhar('adicionar')
        self.entregues.append((kit_id, item.nome, item.quantidade))
        self.itens.setdefault(kit_id, []).append((100 + len(self.entregues), item.nome, item.quantidade))

    def remover(self, item_id):
        self._talvez_falhar('remover')
        self.removidos.append(item_id)
        for kit_id, itens in self.itens.items():
            self.itens[kit_id] = [i for i in itens if i[0] != item_id]

    def editar(self, item_id, antes, depois):
        self._talvez_falhar('editar')
        for itens in self.itens.values():
            for i, (atual_id, nome, quantidade) in enumerate(itens):
                if atual_id == item_id:
                    if (nome, quantidade) != (antes.nome, antes.quantidade):
                        raise ValueError(f'{antes.nome} mudou depois que você leu: agora é {nome} com {quantidade}')
                    itens[i] = (item_id, depois.nome, depois.quantidade)
                    self.editados.append((item_id, depois.nome, depois.quantidade))
                    return
        raise LookupError(f'Item {item_id} não existe')

    def remover_kit(self, kit_id):
        self._talvez_falhar('remover_kit')
        if self.itens.get(kit_id):
            raise ValueError(f'Kit {kit_id} ainda tem itens: remova os itens antes')
        self.kits = [k for k in self.kits if k[0] != kit_id]
        self.kits_removidos.append(kit_id)

    def perguntar(self, texto):
        self.perguntas.append(texto)
        return self.resposta

    def pedir_nome(self, texto):
        self.pedidos_de_nome.append(texto)
        return self.nome_novo


class TestJanela(unittest.TestCase):
    def abrir(self, banco=None, intervalo_ms=60_000):
        self.banco = banco or BancoDeMentira()
        janela = JanelaItens(self.banco, self.banco.perguntar, intervalo_ms, pedir_nome=self.banco.pedir_nome)
        self.addCleanup(fechar, janela)
        janela.update()
        return janela

    def linhas(self, janela):
        return [tuple(janela.lista.item(i, 'values')) for i in janela.lista.get_children()]

    def escolher(self, janela, kit_id):
        rotulo = next(r for r in janela.kit.cget('values') if r.endswith(f'(kit {kit_id})'))
        janela.kit.set(rotulo)
        janela.mostrar()

    def preencher(self, janela, nome, quantidade):
        for campo, texto in ((janela.nome, nome), (janela.quantidade, quantidade)):
            campo.delete(0, 'end')
            if texto:
                campo.insert(0, texto)

    def selecionar(self, janela, item_id):
        janela.lista.selection_set(str(item_id))
        janela.update()

    def erros(self, janela):
        return janela.erro_nome.cget('text'), janela.erro_quantidade.cget('text')

        fim = time.monotonic() + 0.7

        janela = self.abrir()

"""Resumo herdado, por cima da sua janela e do seu DadosMySQL. Uso: python abrir_legado.py"""
from abrir_janela import DadosMySQL, perguntar, pedir_nome
from kits.banco import conexao_banco
from kits.janela import JanelaItens


class DadosLegados(DadosMySQL):
    def resumo_kits(self):
        with conexao_banco(self.config) as con:
            cur = con.cursor()
            cur.execute('SELECT MIN(k.id), k.nome, COUNT(*), SUM(IFNULL(i.quantidade, 0)) '
                        'FROM kit k LEFT JOIN item i ON i.kit_id = k.id GROUP BY k.nome ORDER BY k.nome')
            return [(kit_id, nome, itens, int(unidades)) for kit_id, nome, itens, unidades in cur.fetchall()]

    def itens_abaixo(self, limite):
        with conexao_banco(self.config) as con:
            cur = con.cursor()
            cur.execute('SELECT k.nome, i.nome, i.quantidade FROM item i, kit k '
                        'WHERE i.quantidade <= %s ORDER BY i.quantidade, k.nome', (limite,))
            return cur.fetchall()
